import { SidebarProvider } from '@preview/shell/components/ui/sidebar';
import { type CSSProperties, useEffect, useId, useRef, useState } from 'react';

import { skin } from './config';
import { PARAMS_MESSAGE } from './messages';
import { Navbar } from './navbar';
import { OptionsPanel } from './options-panel';
import { readParams, writeParams, type WorkbenchParams } from './params';
import { usePreferences } from './preferences';
import { readStored, writeStored } from './storage';
import { createUrlWriter } from './url';

const writeUrl = createUrlWriter();

/**
 * A slimmed-down Video.js sandbox for one skin: the options panel sets how the preview looks, and the preview frame
 * renders the skin on its own page so the shell's styles never reach it.
 */
export function App() {
  const optionsId = useId();
  const frame = useRef<HTMLIFrameElement>(null);
  const [params, setParams] = useState(() => readParams(location.search));
  const [optionsOpen, setOptionsOpen] = useState(() => readStored('options-open', true));
  // The frame loads once with the first settings; later changes are posted to it, so playback is never interrupted.
  const [frameUrl] = useState(() => `./preview.html#${writeParams(params)}`);
  const preferences = usePreferences();
  const query = writeParams(params);

  // Every setting lives in the URL, so a reload or a shared link reproduces the preview.
  useEffect(() => {
    // The frame updates straight away; the address bar catches up once changes pause.
    writeUrl(`?${query}`);
    frame.current?.contentWindow?.postMessage({ type: PARAMS_MESSAGE, query }, location.origin);
  }, [query]);

  // The scheme control drives the shell too, so its chrome matches what the preview shows.
  useEffect(() => {
    if (params.scheme === 'auto') delete document.documentElement.dataset.colorScheme;
    else document.documentElement.dataset.colorScheme = params.scheme;
  }, [params.scheme]);

  const update = (value: Partial<WorkbenchParams>) => setParams((current) => ({ ...current, ...value }));

  return (
    <SidebarProvider
      open={optionsOpen}
      onOpenChange={(open) => {
        writeStored('options-open', open);
        setOptionsOpen(open);
      }}
      className="bg-background text-foreground h-screen min-h-0 flex-col overflow-hidden"
      style={{ '--sidebar-width': '20rem' } as CSSProperties}
    >
      <Navbar previewUrl={`./preview.html#${query}`} optionsId={optionsId} />
      <div className="flex min-h-0 flex-1">
        <main className="flex min-h-0 min-w-0 flex-1 flex-col">
          <iframe
            ref={frame}
            src={frameUrl}
            onLoad={() => frame.current?.contentWindow?.postMessage({ type: PARAMS_MESSAGE, query }, location.origin)}
            className="min-h-0 w-full flex-1 border-0"
            title={`${skin.name} preview`}
          />
        </main>
        <OptionsPanel id={optionsId} params={params} onParamsChange={update} preferences={preferences} />
      </div>
    </SidebarProvider>
  );
}

import { ArrowTopRightOnSquareIcon, BookOpenIcon, Cog8ToothIcon } from '@heroicons/react/16/solid';
import { Badge } from '@preview/shell/components/ui/badge';
import { Button } from '@preview/shell/components/ui/button';
import { useSidebar } from '@preview/shell/components/ui/sidebar';

import { skin } from './config';

type NavbarProps = {
  /** The preview on its own, to open in a new tab. */
  previewUrl: string;
  /** The options panel this bar's toggle opens and closes. */
  optionsId: string;
};

/** The skin and links out. Everything else sits in the options panel. */
export function Navbar({ previewUrl, optionsId }: NavbarProps) {
  const { isMobile, open, openMobile, toggleSidebar } = useSidebar();

  return (
    <header className="border-border bg-background flex h-14 shrink-0 items-center gap-4 border-b px-4">
      <div className="flex min-w-0 items-center gap-2">
        <svg viewBox="0 0 209 209" width="18" height="18" aria-hidden="true" xmlns="http://www.w3.org/2000/svg">
          <path fill="#ffa81b" d="M0 0h209v209H0z" />
          <path fill="#ff6200" d="M0 80h209v129H0z" />
          <path fill="#eb3132" d="M0 142h209v67H0z" />
          <path fill="#cc3566" d="M0 187h209v22H0z" />
          <path d="M167.09 104.5L65.13 169.81V39.19l101.96 65.31z" fill="#ebe4c1" />
        </svg>
        <h1 className="text-foreground truncate text-base font-semibold tracking-tight">{skin.name}</h1>
        <Badge variant="outline">
          {skin.frameworkLabel} Â· {skin.presetLabel}
        </Badge>
      </div>

      <div className="ml-auto flex items-center gap-1">
        <Button
          variant="ghost"
          size="icon-sm"
          nativeButton={false}
          render={
            <a href={previewUrl} target="_blank" rel="noopener noreferrer" aria-label="Open the preview in a new tab" />
          }
        >
          <ArrowTopRightOnSquareIcon className="size-4" aria-hidden="true" />
        </Button>
        <Button
          variant="ghost"
          size="icon-sm"
          nativeButton={false}
          render={
            <a
              href={skin.docsUrl}
              target="_blank"
              rel="noopener noreferrer"
              aria-label="Build your own skin guide (opens in a new tab)"
            />
          }
        >
          <BookOpenIcon className="size-4" aria-hidden="true" />
        </Button>
        <Button
          id={`${optionsId}-trigger`}
          type="button"
          variant="ghost"
          size="icon-sm"
          aria-label="Options"
          aria-expanded={isMobile ? openMobile : open}
          aria-controls={optionsId}
          onClick={() => {
            toggleSidebar();

            if (!isMobile && !open) {
              requestAnimationFrame(() => document.getElementById(`${optionsId}-close`)?.focus());
            }
          }}
        >
          <Cog8ToothIcon className="size-4" aria-hidden="true" />
        </Button>
      </div>
    </header>
  );
}

import { Slider as SliderPrimitive } from '@base-ui/react/slider';
import { cn } from 'cn';

function Slider({
  className,
  'aria-label': label,
  'aria-labelledby': labelledBy,
  'aria-valuetext': valueText,
  defaultValue,
  value,
  min = 0,
  max = 100,
  ...props
}: SliderPrimitive.Root.Props) {
  const values = value ?? defaultValue ?? min;
  const count = Array.isArray(values) ? values.length : 1;

  return (
    <SliderPrimitive.Root
      className={cn('data-[orientation=horizontal]:w-full data-[orientation=vertical]:h-full', className)}
      data-slot="slider"
      defaultValue={defaultValue}
      value={value}
      min={min}
      max={max}
      thumbAlignment="edge"
      {...props}
    >
      <SliderPrimitive.Control className="relative flex h-full w-full touch-none items-center select-none data-disabled:opacity-50 data-[orientation=vertical]:h-full data-[orientation=vertical]:min-h-40 data-[orientation=vertical]:w-auto data-[orientation=vertical]:flex-col">
        <SliderPrimitive.Track
          data-slot="slider-track"
          className="bg-muted relative grow overflow-hidden rounded-full select-none data-[orientation=horizontal]:h-1 data-[orientation=horizontal]:w-full data-[orientation=vertical]:h-full data-[orientation=vertical]:w-1"
        >
          <SliderPrimitive.Indicator
            data-slot="slider-range"
            className="bg-primary select-none data-[orientation=horizontal]:h-full data-[orientation=vertical]:w-full"
          />
        </SliderPrimitive.Track>
        {Array.from({ length: count }, (_, index) => (
          <SliderPrimitive.Thumb
            data-slot="slider-thumb"
            key={index}
            aria-label={label}
            aria-labelledby={labelledBy}
            aria-valuetext={valueText}
            className="border-primary/30 hover:border-primary/60 focus-visible:border-primary/60 active:border-primary/60 ring-ring/50 relative block size-3.5 shrink-0 rounded-full border bg-white bg-clip-padding transition-[color,box-shadow] select-none after:absolute after:-inset-2 hover:ring-3 focus-visible:ring-3 focus-visible:outline-hidden active:ring-3 disabled:pointer-events-none disabled:opacity-50"
          />
        ))}
      </SliderPrimitive.Control>
    </SliderPrimitive.Root>
  );
}

export { Slider };
